21:     /// - Parameters:
22:     ///   - newKeys: New Key tensor.
23:     ///   - newValues: New Value tensor.
24:     ///   - axis: Optional explicit concatenation axis (defaults to 2 for 4D attention tensors, 1 for 2D).
25:     /// - Returns: Accumulated (Key, Value) tensors over all past steps.

27:         keys newKeys: MLXArray,
28:         values newValues: MLXArray,
29:         axis: Int? = nil

31:         let concatAxis = axis ?? (newKeys.ndim == 4 ? 2 : 1)

5: /// A dynamic paged Key-Value attention cache allocator for high-throughput batch and stream inference.
6: ///
7: /// `PagedKVCache` organizes Key-Value tensor storage into fixed-size physical memory pages (blocks),
8: /// eliminating frequent buffer re-allocations and memory fragmentation in unified memory.
9: public final class PagedKVCache: @unchecked Sendable {
10:     /// Number of token slots per page block.
11:     public let pageSize: Int
12:     /// Dimensionality of each attention head.
13:     public let headDim: Int
14:     /// Number of attention heads.
15:     public let numHeads: Int
16:     /// Number of transformer layers.
17:     public let numLayers: Int
18: 
19:     private var physicalKeyPages: [[[MLXArray]]] // [layer][pageIdx][slotIdx]
20:     private var physicalValuePages: [[[MLXArray]]] // [layer][pageIdx][slotIdx]
21:     private var sequenceBlockTables: [Int: [Int]] // sequenceId -> [pageIdx]
22:     private var sequenceLengths: [Int: Int] // sequenceId -> length
23: 
24:     /// Initializes a paged KV-cache allocator.
25:     ///
26:     /// - Parameters:
27:     ///   - pageSize: Number of tokens per page block (default: 16).
28:     ///   - numHeads: Number of attention heads.
29:     ///   - headDim: Dimension of each attention head.
30:     ///   - numLayers: Number of transformer layers.
31:     public init(
32:         pageSize: Int = 16,
33:         numHeads: Int = 4,
34:         headDim: Int = 32,
35:         numLayers: Int = 1
36:     ) {
37:         self.pageSize = pageSize
38:         self.numHeads = numHeads
39:         self.headDim = headDim
40:         self.numLayers = numLayers
41: 
42:         self.physicalKeyPages = Array(repeating: [], count: numLayers)
43:         self.physicalValuePages = Array(repeating: [], count: numLayers)
44:         self.sequenceBlockTables = [:]
45:         self.sequenceLengths = [:]
46:     }
47: 
48:     /// Appends single-step Key and Value tokens for a sequence and layer into the paged memory pool.
49:     ///
50:     /// - Parameters:
51:     ///   - sequenceId: Identifier of the sequence.
52:     ///   - layer: The layer index.
53:     ///   - key: New Key tensor of shape `[1, 1, numHeads, headDim]` or `[numHeads, headDim]`.
54:     ///   - value: New Value tensor of shape `[1, 1, numHeads, headDim]` or `[numHeads, headDim]`.
55:     public func append(sequenceId: Int, layer: Int, key: MLXArray, value: MLXArray) {
56:         let currentLen = sequenceLengths[sequenceId, default: 0]
57:         var blockTable = sequenceBlockTables[sequenceId, default: []]
58: 
59:         let pageIndexInSeq = currentLen / pageSize
60:         let offsetInPage = currentLen % pageSize
61: 
62:         if pageIndexInSeq >= blockTable.count {
63:             // Allocate new physical page for all layers
64:             let newPageIdx = physicalKeyPages[layer].count
65:             let emptyPageK = (0..<pageSize).map { _ in MLX.zeros([numHeads, headDim]) }
66:             let emptyPageV = (0..<pageSize).map { _ in MLX.zeros([numHeads, headDim]) }
67:             physicalKeyPages[layer].append(emptyPageK)
68:             physicalValuePages[layer].append(emptyPageV)
69:             blockTable.append(newPageIdx)
70:             sequenceBlockTables[sequenceId] = blockTable
71:         }
72: 
73:         let physicalPageIdx = blockTable[pageIndexInSeq]
74:         let reshapedK = key.reshaped([numHeads, headDim])
75:         let reshapedV = value.reshaped([numHeads, headDim])
76: 
77:         physicalKeyPages[layer][physicalPageIdx][offsetInPage] = reshapedK
78:         physicalValuePages[layer][physicalPageIdx][offsetInPage] = reshapedV
79: 
80:         if layer == numLayers - 1 {
81:             sequenceLengths[sequenceId] = currentLen + 1
82:         }
83:     }
84: 
85:     /// Retrieves concatenated Keys and Values for all past tokens of a sequence at a specified layer.
86:     ///
87:     /// - Parameters:
88:     ///   - sequenceId: Identifier of the sequence.
89:     ///   - layer: The layer index.
90:     /// - Returns: Accumulated (Keys, Values) of shape `[1, seqLen, numHeads, headDim]`.
91:     public func getKeysAndValues(sequenceId: Int, layer: Int) -> (keys: MLXArray, values: MLXArray) {
92:         let totalLen = sequenceLengths[sequenceId, default: 0]
93:         guard totalLen > 0 else {
94:             let empty = MLX.zeros([1, 0, numHeads, headDim])
95:             return (empty, empty)
96:         }
97: 
98:         guard let blockTable = sequenceBlockTables[sequenceId], !blockTable.isEmpty else {
99:             let empty = MLX.zeros([1, 0, numHeads, headDim])
100:             return (empty, empty)
101:         }
102: 
103:         var keySlots: [MLXArray] = []
104:         var valSlots: [MLXArray] = []
105: 
106:         var remaining = totalLen
107:         for physIdx in blockTable {
108:             let countInThisPage = min(remaining, pageSize)
109:             for slot in 0..<countInThisPage {
110:                 keySlots.append(physicalKeyPages[layer][physIdx][slot])
111:                 valSlots.append(physicalValuePages[layer][physIdx][slot])
112:             }
113:             remaining -= countInThisPage
114:             if remaining <= 0 { break }
115:         }
116: 
117:         let keys = MLX.stacked(keySlots, axis: 0).expandedDimensions(axis: 0)
118:         let vals = MLX.stacked(valSlots, axis: 0).expandedDimensions(axis: 0)
119:         return (keys, vals)
120:     }
121: 
122:     /// Resets the memory allocation table for a sequence.
123:     ///
124:     /// - Parameter sequenceId: Identifier of the sequence to free.
125:     public func reset(sequenceId: Int) {
126:         sequenceBlockTables.removeValue(forKey: sequenceId)
127:         sequenceLengths.removeValue(forKey: sequenceId)
128:     }
129: 
130:     /// Total number of physical page blocks allocated across all layers.
131:     public var totalAllocatedPages: Int {
132:         physicalKeyPages.reduce(0) { $0 + $1.count }
133:     }
134: }
